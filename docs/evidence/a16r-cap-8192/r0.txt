class CoalesceRangesTests(unittest.TestCase):
    def test_unsorted_overlap(self):

        # items unchanged

        self.assertEqual(coalesce_ranges([(0,0)], limit=0), [])

        with self.assertRaises(ValueError):
            coalesce_ranges([(0,11)], limit=10)
        with self.assertRaises(ValueError):

from collections.abc import Iterable
import unittest


def _is_exact_int(value: object) -> bool:
    return isinstance(value, int) and not isinstance(value, bool)


def coalesce_ranges(
    ranges: Iterable[tuple[int, int]],
    *,
    limit: int | None = None,
) -> list[tuple[int, int]]:
    """Validate, sort, and merge half-open byte ranges [start, end)."""
    if limit is not None and (not _is_exact_int(limit) or limit < 0):
        raise ValueError("limit must be an exact nonnegative integer")

    validated: list[tuple[int, int]] = []

    for item in ranges:
        if not isinstance(item, tuple) or len(item) != 2:
            raise ValueError("each range must be a 2-tuple (start, end)")

        start, end = item

        if not _is_exact_int(start) or not _is_exact_int(end):
            raise ValueError("start and end must be exact integers")

        if start < 0:
            raise ValueError("start must be nonnegative")

        if start > end:
            raise ValueError("start must be <= end")

        if limit is not None and end > limit:
            raise ValueError("end must be <= limit")

        if start == end:
            continue

        validated.append((start, end))

    validated.sort(key=lambda pair: (pair[0], pair[1]))

    merged: list[tuple[int, int]] = []
    for start, end in validated:
        if not merged:
            merged.append((start, end))
            continue

        prev_start, prev_end = merged[-1]
        if start <= prev_end:
            merged[-1] = (prev_start, max(prev_end, end))
        else:
            merged.append((start, end))

    return merged
