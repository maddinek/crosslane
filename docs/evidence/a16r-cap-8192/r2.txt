from typing import Iterable, Optional

Range = tuple[int, int]

def coalesce_ranges(
    ranges: Iterable[Range],
    *,
    limit: Optional[int] = None,
) -> list[Range]:
    """Coalesce half-open byte ranges [start, end) into sorted, non-overlapping ranges.

    Validates exact integer endpoints, optional limit, discards empty ranges,
    and raises ValueError for malformed or invalid input.
    """
    if limit is not None:
        if type(limit) is not int or limit < 0:
            raise ValueError("limit must be an exact nonnegative integer")
    normalized: list[Range] = []
    for item in ranges:
        try:
            start, end = item
        except (TypeError, ValueError) as exc:
            raise ValueError("each range must be a (start, end) pair") from exc
        # The unpack ensures exactly two elements? Actually unpack start,end = item raises if length not 2 or not iterable.
        if type(start) is not int or type(end) is not int:
            raise ValueError("range endpoints must be exact integers")
        if start < 0 or start > end:
            raise ValueError("range must satisfy 0 <= start <= end")
        if limit is not None and end > limit:
            raise ValueError("range end must be <= limit")
        if start == end:
            continue
        normalized.append((start, end))
    normalized.sort(key=lambda r: (r[0], r[1]))
    merged: list[Range] = []
    for start, end in normalized:
        if not merged:
            merged.append((start, end))
        else:
            prev_start, prev_end = merged[-1]
            if start <= prev_end:
                merged[-1] = (prev_start, max(prev_end, end))
            else:
                merged.append((start, end))
    return merged

) -> list[Range]:
    """Coalesce half-open byte ranges [start, end) into sorted merged ranges.

    Validates exact integer endpoints, discards empty ranges, merges overlapping
    or touching ranges, and raises ValueError for malformed or invalid input.
    """
    if limit is not None:
        if type(limit) is not int or limit < 0:
            raise ValueError("limit must be an exact nonnegative integer")

    normalized: list[Range] = []

    for item in ranges:
        try:
            start, end = item
        except (TypeError, ValueError) as exc:
            raise ValueError("each range must be a (start, end) pair") from exc

        if type(start) is not int or type(end) is not int:
            raise ValueError("range endpoints must be exact integers")

        if start < 0 or start > end:
            raise ValueError("range must satisfy 0 <= start <= end")

        if limit is not None and end > limit:
            raise ValueError("range end must be <= limit")

        if start == end:
            continue

        normalized.append((start, end))

    normalized.sort(key=lambda r: (r[0], r[1]))

    merged: list[Range] = []

    for start, end in normalized:
        if not merged:
            merged.append((start, end))
        else:
            previous_start, previous_end = merged[-1]
            if start <= previous_end:
                merged[-1] = (previous_start, max(previous_end, end))
            else:
                merged.append((start, end))

    return merged
