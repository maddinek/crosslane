from __future__ import annotations? Python 3.11. Fine.
from typing import Iterable
import unittest

def _is_exact_int(value: object) -> bool:
    return isinstance(value, int) and not isinstance(value, bool)

def coalesce_ranges(ranges: Iterable[tuple[int, int]], *, limit: int | None = None) -> list[tuple[int, int]]:
    """
    Coalesce half-open byte ranges [start, end) into sorted, merged tuples.

    The input iterable is consumed exactly once. Empty ranges (start == end)
    are validated and discarded. Overlapping or touching ranges are merged.
    Raises ValueError for invalid endpoints, malformed pairs, or a bad limit.
    """
    if limit is not None and (not _is_exact_int(limit) or limit < 0):
        raise ValueError("limit must be an exact nonnegative integer")

    valid: list[tuple[int, int]] = []

    for item in ranges:
        try:
            start, end = item
        except (TypeError, ValueError) as exc:
            raise ValueError("each range must be a 2-item pair") from exc

        if not _is_exact_int(start) or not _is_exact_int(end):
            raise ValueError("range endpoints must be exact integers")
        if start < 0 or end < start:
            raise ValueError("range endpoints must satisfy 0 <= start <= end")
        if limit is not None and end > limit:
            raise ValueError("range end exceeds limit")

        if start == end:
            continue
        valid.append((start, end))

    valid.sort(key=lambda r: r[0])

    merged: list[tuple[int, int]] = []
    for start, end in valid:
        if merged and start <= merged[-1][1]:
            prev_start, prev_end = merged[-1]
            if end > prev_end:
                merged[-1] = (prev_start, end)
        else:
            merged.append((start, end))

    return merged
